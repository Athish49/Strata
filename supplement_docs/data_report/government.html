Stitcher logic summary [code]

FR final_rule / correction / direct_final_rule / interim_final_rule → CFR via cfr_references (exact then prefix, 365-day window). Reverse: CFR → FR (30-day window). State (IURC/IDEM) → IAC via iac_cross_refs in legal_refs, or DIN fragment match: %-IR-{TITLE3D}{YEAR2D}{LSA4D}%.

DIN encoding: YYYYMMDD-IR-{title:03d}{year2d}{lsa:04d}{type}. Titles 610/675 DIN matching produces 0 hits despite 90/954 sections having DINs.

§12 Join Guide

A: Code section changed between snapshots

-- Join A: find changed sections (S1 vs S2)
SELECT cs2.id, cs1.id AS s1_id, cs2.citation,
       cs1.content_hash AS s1_hash, cs2.content_hash AS s2_hash,
       cs2.owning_agency, cs2.heading
FROM code_sections cs2
JOIN code_sections cs1 ON cs1.citation = cs2.citation
  AND cs1.source_system = cs2.source_system
  AND cs1.snapshot_date < cs2.snapshot_date
WHERE cs2.content_hash != cs1.content_hash
  AND cs2.source_system = 'cfr';

Runtime from live test: ~46ms. Returns 45 CFR changed pairs. [live]

B: Fetch latest snapshot per citation

-- Join B: latest IAC sections (snapshot_date DESC)
SELECT id, citation, source_system, snapshot_date, status, heading, content_hash
FROM code_sections
WHERE source_system = 'iac'
  AND snapshot_date = (
    SELECT MAX(snapshot_date) FROM code_sections WHERE source_system = 'iac'
  )
ORDER BY citation;

Runtime from live test: ~45ms. [live]

C: Code section → amending action

-- Join C: code section with its amending FR action
SELECT cs.id, cs.citation, cs.snapshot_date, cs.amendment_source,
       ra.source_id, ra.title, ra.date_published, ra.action_type, ra.agency
FROM code_sections cs
JOIN regulatory_actions ra ON ra.source_id = cs.amendment_source
WHERE cs.source_system = 'cfr'
  AND cs.snapshot_date = '2026-10-02'
LIMIT 10;

Neon: code_sections stats by source_system 
SELECT source_system, jurisdiction_level, jurisdiction_geo,
  COUNT(*) AS cnt, COUNT(DISTINCT citation) AS distinct_citations,
  COUNT(DISTINCT snapshot_date) AS snapshot_count,
  MIN(snapshot_date) AS earliest_snapshot, MAX(snapshot_date) AS latest_snapshot,
  COUNT(*) FILTER (WHERE body_text IS NULL OR length(body_text) = 0) AS empty_body,
  COUNT(*) FILTER (WHERE amendment_source IS NOT NULL) AS has_amendment_source,
  COUNT(*) FILTER (WHERE repealed_date IS NOT NULL) AS repealed,
  ROUND(AVG(length(body_text))) AS avg_body_chars,
  PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY length(body_text)) AS p95_body_chars,
  MAX(length(body_text)) AS max_body_chars
FROM code_sections
GROUP BY source_system, jurisdiction_level, jurisdiction_geo
ORDER BY source_system

Neon: snapshot diff (changed/unchanged/new/removed) 
WITH snaps AS (
  SELECT source_system, MIN(snapshot_date) AS s1, MAX(snapshot_date) AS s2
  FROM code_sections
  WHERE source_system IN (
    SELECT source_system FROM code_sections
    GROUP BY source_system HAVING COUNT(DISTINCT snapshot_date) >= 2
  )
  GROUP BY source_system
)
SELECT s.source_system, s.s1, s.s2,
  COUNT(*) FILTER (WHERE cs1.id IS NOT NULL AND cs2.id IS NOT NULL AND cs1.content_hash != cs2.content_hash) AS changed,
  COUNT(*) FILTER (WHERE cs1.id IS NOT NULL AND cs2.id IS NULL) AS removed_in_s2
FROM snaps s
LEFT JOIN code_sections cs1 ON cs1.source_system = s.source_system AND cs1.snapshot_date = s.s1
LEFT JOIN code_sections cs2 ON cs2.source_system = s.source_system AND cs2.snapshot_date = s.s2
  AND cs2.citation = cs1.citation
GROUP BY s.source_system, s.s1, s.s2
